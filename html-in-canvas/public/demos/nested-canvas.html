tv Nested Canvas (HTML-in-Canvas)

Demonstrates nested canvas rendering where all drawn elements are direct children of their respective canvas. The outer canvas (#tv-frame) draws its direct children: #tv-casing, #tv-screen (inner canvas), and #power-btn. The inner canvas (#tv-screen) draws its direct children: #movie-player and #off-state-black-screen.

<!-- Outer Canvas contains direct children on which drawElementImage is invoked -->
<canvas id="tv-frame" layoutsubtree>
  <!-- Direct Child 1 -->
  <div id="tv-casing" drawable></div>

  <!-- Direct Child 2: Inner Canvas -->
  <canvas id="tv-screen" drawable layoutsubtree>
    <!-- Direct Children of Inner Canvas -->
    <video id="movie-player" drawable src="../assets/demo-video.mp4" style="opacity: 0.7;"></video>
    <div id="off-state-black-screen" drawable></div>
  </canvas>

  <!-- Direct Child 3: Power Button -->
  <button id="power-btn" drawable onclick="togglePower()">Turn TV On</button>
</canvas>

<script>
  function renderFrame() {
    // 1. Outer Canvas draws its direct children: tvCasing and powerBtn
    outerCtx.drawElementImage(tvCasing, casingX, casingY);
    const btnTransform = outerCtx.drawElementImage(powerBtn, btnX, btnY);
    powerBtn.style.transform = btnTransform.toString();

    // 2. Inner Canvas draws its direct children: tvNoise + video (or blackScreen)
    if (isTvOn) {
      drawTvNoise(innerCtx, innerCanvas.width, innerCanvas.height);
      innerCtx.globalAlpha = 0.7;
      innerCtx.drawElementImage(video, 0, 0, innerCanvas.width, innerCanvas.height);
    }

    // 3. Outer Canvas draws its direct child: innerCanvas
    const screenTransform = outerCtx.drawElementImage(innerCanvas, screenX, screenY);
    innerCanvas.style.transform = screenTransform.toString();

    requestAnimationFrame(renderFrame);
  }
</script>
